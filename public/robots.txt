# The Larkin is a demonstration property. It does not exist.
#
# Indexing it would put a fictional building with fictional rents into search
# results next to real listings, which is how a demo turns into a complaint.
User-agent: *
Disallow: /

# The operations routes are named explicitly as well. Not because a crawler that
# honours the blanket rule above needs telling twice, but because the day someone
# opens the site to indexing, the staff routes must not come along. robots.txt is a
# request, never a control — server authentication and property authorization keep
# prospect names and call excerpts private.
Disallow: /api/
Disallow: /api/dashboard
Disallow: /api/properties
Disallow: /dashboard.html
